Detailed Description

Utility class for inspecting and debugging Protobuf MaterialXDocument messages.

  • Print the protobuf document structure in various formats.
  • Generate Mermaid diagrams for visualization of the document hierarchy.

298 def generate_mermaid_diagram(pb_doc):
299 """
300 @brief Generate a Mermaid diagram (graph LR) from a protobuf document hierarchy.
301 @param pb_doc Protobuf message describing the document
302 @return str Mermaid diagram source code.
303 """
304 data = MessageToDict(pb_doc)
305 mermaid_lines = ["graph LR"]
306
307 def add_mermaid_elements(element):
308 node_id = element.get('name').replace(' ', '_')
309 label = f"{element.get('name')} : {element.get('type')}"
310 mermaid_lines.append(f" {node_id}[{label}]")
311
312 for child in element.get('children', []):
313 child_id = child.get('name').replace(' ', '_')
314 add_mermaid_elements(child)
315 mermaid_lines.append(f" {node_id} --> {child_id}")
316
317 for element in data.get('elements', []):
318 add_mermaid_elements(element)
319
320 mermaid_code = "\n".join(mermaid_lines)
321
322 return mermaid_code
